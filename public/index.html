// app/projects/[id]/page.tsx
import { createClient } from '@/lib/supabase'
import { Timeline } from '@/components/timeline'

export default async function ProjectPage({ params }) {
  const db = createClient()
  const { data: project } = await db
    .from('projects')
    .select('*, stages(*), payments(*)')
    .eq('id', params.id)
    .single()

  return (
    <main dir="rtl">
      <Timeline steps={project.stages} />
      <Payments items={project.payments} />
    </main>
  )
}

export async function POST(req: Request) {

  const session = await pay.create({
    amount: plan.price, currency: 'SAR',
    methods: ['mada', 'applepay', 'visa'],
  })
  return Response.json({ url: session.url })
}

أسئلة شائعة

أكثر ما يسأل عنه العملاء قبل بدء المشروع.

كم تستغرق برمجة موقع أو منصة ؟

يعتمد على حجم المشروع: موقع تعريفي من 5 إلى 10 أيام عمل، متجر إلكتروني أو منصة تعليمية من 3 إلى 6 أسابيع، وتطبيق جوال من 6 إلى 10 أسابيع. نحدد الجدول الزمني بدقة في العرض الفني قبل البدء.

ما التقنيات المستخدمة في التطوير ؟

Next.js وReact وTypeScript للواجهات، وNode.js وSupabase وPostgreSQL للخلفية، وSwift وSwiftUI لتطبيقات iOS. كلها تقنيات حديثة ومدعومة طويل الأمد، فمشروعك يظل قابلًا للتطوير لسنوات.

هل أملك كود المشروع بعد التسليم ؟

نعم. جميع الأكواد المصدرية وملفات المشروع ملك كامل لصاحب المشروع فور سداد كامل قيمة الباقة، مع تسليم كل الصلاحيات والوصول — ولا يحق لنا إعادة استخدامها أو بيعها لطرف آخر.

هل الموقع يظهر في نتائج جوجل ؟

نعم. كل مشروع يُسلَّم بتحسين سيو أساسي: وسوم العنوان والوصف، بيانات منظمة Schema، خريطة موقع، بنية عناوين دلالية، وسرعة تحميل عالية — وهي أساس الظهور في نتائج البحث.

كيف أتابع مشروعي والدفعات ؟

لك بوابة خاصة على المنصة تتابع منها مراحل التنفيذ لحظة بلحظة، وحالة كل دفعة، والملفات المسلَّمة — بدل رسائل متفرقة وملفات ضائعة.

ما وسائل الدفع المتاحة ؟

فيزا وماستركارد وApple Pay، وإنستا باي وفودافون كاش وأورنج كاش داخل مصر. والسداد على دفعات مرتبطة بمراحل التنفيذ — لا دفعة قبل إنجاز مرحلتها.
